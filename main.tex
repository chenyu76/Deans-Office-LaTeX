\def\currentSidebarState{0}% 当前页输出时使用的状态（0=关，1=开）
\def\pendingSidebarState{0}% 用户设定的待生效状态
\newcommand{\EnableSidebar}{%
  \global\def\pendingSidebarState{1}%
  % 注意：不修改 \currentSidebarState
}
\newcommand{\DisableSidebar}{%
  \global\def\pendingSidebarState{0}%
  \global\def\currentSidebarState{0}% 强制当前页输出时不画
}

% 页面外框

  \global\def\pendingPageFrameState{1}%
}
\newcommand{\DisablePageFrame}{%
  \global\def\pendingPageFrameState{0}%
  \global\def\currentPageFrameState{0}%
}

% 绘制全页外框 (Hook到每页输出)
\newcommand{\DrawFullPageFrame}{%
  \begin{tikzpicture}[overlay, remember picture]
    \draw
    ([xshift=\PageMargin, yshift=-\PageMargin]current page.north west)
    rectangle
    ([xshift=-\PageMargin, yshift=\PageMargin]current page.south east);
  \end{tikzpicture}%
}

% 页面外框首半帧（从当前位置向下画到页底，加顶横线）
\newcommand{\DrawPageFrameHead}{%
  \begin{tikzpicture}[overlay, remember picture]
    \path (current page.west);
    \pgfgetlastxy{\westx}{\dummy};
    \path (current page.east);
    \pgfgetlastxy{\eastx}{\dummyy};
    \path (0,0);
    \pgfgetlastxy{\dummyx}{\currenty}
    \path (current page.south);
    \pgfgetlastxy{\dummyx}{\southy}
    \pgfmathsetlengthmacro{\m}{\PageMargin}
    \draw
    ({\westx + \m}, 0) rectangle ({\eastx - \m}, {\southy - \currenty + \m});
  \end{tikzpicture}%
}

% 页面外框末半帧（从当前位置向上画到页顶，加底横线封口）
\newcommand{\DrawPageFrameTail}{%
  \begin{tikzpicture}[overlay, remember picture]
    \path (current page.west);
    \pgfgetlastxy{\westx}{\dummyy};
    \path (current page.east);
    \pgfgetlastxy{\eastx}{\dummmyy};
    \path (0,0);
    \pgfgetlastxy{\dummyx}{\currenty}
    \path (current page.north);
    \pgfgetlastxy{\dummyx}{\northy}
    \pgfmathsetlengthmacro{\m}{\PageMargin}
    \draw
    ({\westx + \m}, 0) rectangle ({\eastx - \m}, {\northy + \currenty - \m});
  \end{tikzpicture}%
}

\newcounter{PageEnvCounter}

% 环境：带框的表格页面（单页）
% 内容必须放在一页内，画完整闭合框
\NewEnviron{TablePage}{%
  \par
  \noindent
  \stepcounter{PageEnvCounter}%
  \setlength{\parindent}{0pt}
  \setlength{\parskip}{0pt}
  \begin{tikzpicture}[overlay, remember picture]
    \path (current page.west);
    \pgfgetlastxy{\westx}{\dummy};
    \path (current page.east);
    \pgfgetlastxy{\eastx}{\dummyy};
    \coordinate (pageStart\thePageEnvCounter) at (0,0);
  \end{tikzpicture}%
  \par
  \nointerlineskip
  \BODY
  \par
  \nointerlineskip
  \noindent
  \begin{tikzpicture}[overlay, remember picture]
    \path (current page.west);
    \pgfgetlastxy{\westx}{\dummyy};
    \path (current page.east);
    \pgfgetlastxy{\eastx}{\dummmyy};
    \pgfmathsetlengthmacro{\m}{\PageMargin}
    \coordinate (pageEnd\thePageEnvCounter) at (0,0);
    \path (pageStart\thePageEnvCounter);
    \pgfgetlastxy{\dummyx}{\startY}
    \path (pageEnd\thePageEnvCounter);
    \pgfgetlastxy{\endX}{\endY}
    \draw
    ({\westx + \m}, \startY) rectangle ({\eastx - \m}, \endY);
  \end{tikzpicture}%
  \par
  \nointerlineskip
  \noindent
}

% 环境：带框的表格页面（多页）
% 首半帧→中间页全帧→末半帧，三段式
\NewEnviron{TablePages}{%
  \par
  \noindent
  \setlength{\parindent}{0pt}
  \setlength{\parskip}{0pt}
  \DrawPageFrameHead
  \EnablePageFrame
  \par
  \nointerlineskip
  \BODY
  \par
  \nointerlineskip
  \noindent
  \DisablePageFrame
  \DrawPageFrameTail
  \par
}

% 命令：绘制横向分割线
\newcommand{\SepLine}{%
  \par
  \nointerlineskip % 防止垂直间距干扰
  \noindent
  \begin{tikzpicture}[overlay, remember picture]
    \pgfmathsetlengthmacro{\m}{\PageMargin};
    % \pgfmathsetmacro{\h}{-1};
    % \path (current page.center);
    % \pgfgetlastxy{\centerx}{\centery}; % 提取 x 和 y 坐标
    \path (current page.west);
    \pgfgetlastxy{\westx}{\westy};
    \path (current page.east);
    \pgfgetlastxy{\eastx}{\easty};
    \draw ({\westx + \m}, 0) -- ({\eastx - \m}, 0);
  \end{tikzpicture}
  \par
  \nointerlineskip
}

% 绘制带有垂直间距的分隔线
\newcommand{\vspaceSepLine}{
  \vspace{0.5em}

  \SepLine

  \vspace{0.5em}
}

% 命令：多列行
% 用法: \TableRow[最小高度]{列格式}{内容1 & 内容2...}
% 示例: \TableRow{X|c|X}{ 左边 & 中间 & 右边 }
% 注意：列格式不需要写两边的竖线，系统会自动处理
\NewDocumentCommand{\TableRow}{O{\RowHeight} m m}{%
  \noindent
  \SepLine % 一条分割线
  % 使用 tabularx 占满行宽
  % @{} 消除左右默认间距，使表格紧贴版心边缘
  \begin{tabularx}{\textwidth}{@{} #2 @{}}
    % 在每个单元格内插入一个不可见的rule撑开高度
    \rule{0pt}{#1} #3
  \end{tabularx}%
  \par
  \nointerlineskip % 紧接着画线或下一行
  \SepLine % 每行结束后自动画分割线
}

% 命令：跨页长文本块
% 类似于 Word 的大文本框，可以确定高度
% 用法: \begin{LongTextField}[高度] ... \end{LongTextField}
\NewEnviron{LongTextField}[1][5cm]{%
  \noindent
  \begin{minipage}[t][#1][t]{\textwidth}
    \vspace{0.5em} % 顶部留白
    \BODY
  \end{minipage}
  \par
  \nointerlineskip
  \SepLine
}

% 环境：侧边栏的文本（单页）
% 参数为宽度，侧边栏标题
\NewEnviron{SidebarTablePage}[2]{%
  \par
  \nointerlineskip
  \noindent
  \stepcounter{SidebarEnvCounter}%
  \setlength{\SidebarWidth}{#1}
  \begin{tikzpicture}[overlay, remember picture]
    \path (current page.west);
    \pgfgetlastxy{\westx}{\dummy};
    \coordinate (sidebarStart\theSidebarEnvCounter) at (0,0);
  \end{tikzpicture}%
  \begin{adjustwidth}{#1}{0cm}
    \BODY
  \end{adjustwidth}%
  \begin{tikzpicture}[overlay, remember picture]
    \path (current page.west);
    \pgfgetlastxy{\westx}{\dummyy};
    \path (current page.east);
    \pgfgetlastxy{\eastx}{\dummmyy};
    \coordinate (sidebarEnd\theSidebarEnvCounter) at (0,0);
    \path (sidebarStart\theSidebarEnvCounter);
    \pgfgetlastxy{\dummyx}{\startY}
    \path (sidebarEnd\theSidebarEnvCounter);
    \pgfgetlastxy{\endX}{\endY}
    \pgfmathsetlengthmacro{\w}{\SidebarWidth}
    \pgfmathsetlengthmacro{\m}{\PageMargin}
    \pgfmathsetlengthmacro{\l}{\m + \westx + \w}
    \pgfmathsetlengthmacro{\barcx}{\m + \westx + \w / 2}
    \pgfmathsetlengthmacro{\midY}{(\startY + \endY) / 2}
    \draw (\l, \startY) -- (\l, \endY);
    \draw ({\westx + \m}, \endY) -- ({\eastx - \m}, \endY);
    \node at (\barcx, \midY) {
      \begin{minipage}{#1}
        #2
      \end{minipage}
    };
  \end{tikzpicture}%
  \par
  \nointerlineskip
  \noindent
}
