% 绘制全页外框 (Hook到每页输出)
\newcommand{\DrawFrameOverlay}{%
  \begin{tikzpicture}[overlay, remember picture]
    % 计算坐标：左上角 (L, T) 和 右下角 (R, B)
    % Geometry的坐标原点和TikZ current page有所不同，这里直接用
    % current page偏移计算最稳健
    \draw[line width=\GlobalLineWidth]
    ([xshift=\PageMargin, yshift=-\PageMargin]current page.north west)
    rectangle
    ([xshift=-\PageMargin, yshift=\PageMargin]current page.south east);

    % 如果需要调试，可以打开下面的网格
    % \draw[help lines, step=1cm] (current page.north west) grid
    % (current page.south east);
  \end{tikzpicture}
}
\AtBeginShipout{%
  \AtBeginShipoutAddToBox{%
    % 页面外框
    \ifShowPageFrame
    \DrawFrameOverlay{}%
    \fi
  }%
  % 侧边栏
  % 1. 根据当前状态决定本页是否绘图
  \ifnum\currentSidebarState=1
  \AtBeginShipoutAddToBox{\DrawSidebarVerticalLine}%
  \fi
  % 2. 本页输出完毕后，将待生效状态“转正”给下一页使用
  \xdef\currentSidebarState{\pendingSidebarState}%
}

% 命令：绘制横向分割线
\newcommand{\SepLine}{%
  \par
  \nointerlineskip % 防止垂直间距干扰
  \noindent
  \begin{tikzpicture}[overlay, remember picture]
    \pgfmathsetlengthmacro{\m}{\PageMargin};
    % \pgfmathsetmacro{\h}{-1};
    % \path (current page.center);
    % \pgfgetlastxy{\centerx}{\centery}; % 提取 x 和 y 坐标
    \path (current page.west);
    \pgfgetlastxy{\westx}{\westy};
    \path (current page.east);
    \pgfgetlastxy{\eastx}{\easty};
    \draw ({\westx + \m}, 0) -- ({\eastx - \m}, 0);
  \end{tikzpicture}
  \par
  \nointerlineskip
}

% 命令：多列行
% 用法: \TableRow[最小高度]{列格式}{内容1 & 内容2...}
% 示例: \TableRow{X|c|X}{ 左边 & 中间 & 右边 }
% 注意：列格式不需要写两边的竖线，系统会自动处理
\NewDocumentCommand{\TableRow}{O{\RowHeight} m m}{%
  \noindent
  % 使用 tabularx 占满行宽
  % @{} 消除左右默认间距，使表格紧贴版心边缘
  \begin{tabularx}{\textwidth}{@{} #2 @{}}
    % 在每个单元格内插入一个不可见的rule撑开高度
    \rule{0pt}{#1} #3
  \end{tabularx}%
  \par
  \nointerlineskip % 紧接着画线或下一行
  \SepLine % 每行结束后自动画分割线
}

% 命令：跨页长文本块
% 类似于 Word 的大文本框，可以确定高度
% 用法: \begin{LongTextField}[高度] ... \end{LongTextField}
\NewEnviron{LongTextField}[1][5cm]{%
  \noindent
  \begin{minipage}[t][#1][t]{\textwidth}
    \vspace{0.5em} % 顶部留白
    \BODY
  \end{minipage}
  \par
  \nointerlineskip
  \SepLine
}

% 用于遮住本身已有的框线的一段额外距离
% 需要比线宽宽
\newlength{\smallOverlayLength}
\setlength{\smallOverlayLength}{0.5cm}

\newlength{\HeadingHeight}
% 命令：自定义表头 (覆盖上边框)
% 用法: \PageTitle{标题}{标题高度}
\newcommand{\PageTitle}[2]{%
  \par
  \nointerlineskip
  \noindent
  \setlength{\HeadingHeight}{#2}
  \begin{tikzpicture}[overlay, remember picture]
    % 用白色矩形遮盖顶部的全局边框线
    \fill[white]
    ([xshift=\PageMargin+\GlobalLineWidth-\smallOverlayLength,
    yshift=-\PageMargin+\smallOverlayLength]current page.north west)
    rectangle
    ([xshift=-\PageMargin-\GlobalLineWidth+\smallOverlayLength,
    yshift=-\PageMargin-\HeadingHeight]current page.north east);

    % 绘制标题内容
    \node[yshift=-1.2cm] at ([xshift=0.5\paperwidth,
    yshift=-\PageMargin]current page.north west) {
      \begin{minipage}{\ContentWidth}
        #1
      \end{minipage}
    };

    % 补画标题下方的横线
    \draw[line width=\GlobalLineWidth]
    ([xshift=\PageMargin, yshift=-\PageMargin-\HeadingHeight]current
    page.north west) --
    ([xshift=-\PageMargin, yshift=-\PageMargin-\HeadingHeight]current
    page.north east);
  \end{tikzpicture}
  \vspace{\dimexpr\HeadingHeight-\ContentGap-1em\relax}
  \par
  \nointerlineskip
}

% 命令：创建一段表格之间的空白
% 参数为高度
\newcommand{\BlankBetweenTable}[1]{
  \par
  \nointerlineskip
  \noindent
  \begin{tikzpicture}[overlay, remember picture]
    \pgfmathsetlengthmacro{\h}{-(#1)}
    \pgfmathsetlengthmacro{\m}{\PageMargin};
    \pgfmathsetlengthmacro{\l}{\smallOverlayLength};
    \path (current page.west);
    \pgfgetlastxy{\westx}{\westy};
    \path (current page.east);
    \pgfgetlastxy{\eastx}{\easty};
    \fill[white]({\westx + \m - \l}, 0) rectangle ({\eastx - \m + \l}, \h);
    \draw ({\westx + \m}, 0) -- ({\eastx - \m}, 0);
    \draw ({\westx + \m}, \h) -- ({\eastx - \m}, \h);
  \end{tikzpicture}
  \par
  \nointerlineskip
}

% 环境：侧边栏的文本
% 参数为高度
\NewEnviron{SidebarTableMultiPages}[2]{%
  \par
  \nointerlineskip
  \noindent
  \setlength{\SidebarWidth}{#1}
  \begin{tikzpicture}[overlay, remember picture]
    \path (current page.west);
    \pgfgetlastxy{\westx}{\dummy};
    \path (0,0);
    \pgfgetlastxy{\dummyx}{\currenty}
    \path (current page.south);
    \pgfgetlastxy{\dummyx}{\southy}
    \pgfmathsetlengthmacro{\w}{\SidebarWidth}
    \pgfmathsetlengthmacro{\m}{\PageMargin}
    \pgfmathsetlengthmacro{\l}{\m + \westx + \w}
    \pgfmathsetlengthmacro{\barcx}{\m + \westx + \w / 2}
    \pgfmathsetlengthmacro{\barcy}{(\southy - \currenty + \m)/2}
    \draw (\l, 0) -- (\l, {\southy - \currenty + \m});
    \node at (\barcx, \barcy) {
      \begin{minipage}{#1}
        #2
      \end{minipage}
    };
  \end{tikzpicture}%
  \EnableSidebar%
  \begin{adjustwidth}{#1}{0cm}
    \BODY
  \end{adjustwidth}%
  \DisableSidebar%
  \begin{tikzpicture}[overlay, remember picture]
    \path (current page.west);
    \pgfgetlastxy{\westx}{\dummyy};
    \path (0,0);
    \pgfgetlastxy{\dummyx}{\currenty}
    \path (current page.north);
    \pgfgetlastxy{\dummyx}{\northy}
    \path (current page.east);
    \pgfgetlastxy{\eastx}{\dummmyy};
    \pgfmathsetlengthmacro{\w}{\SidebarWidth}
    \pgfmathsetlengthmacro{\m}{\PageMargin}
    \pgfmathsetlengthmacro{\l}{\m + \westx + \w}
    \draw (\l, 0) -- (\l, {\northy + \currenty - \m});
    \draw ({\westx + \m}, 0) -- ({\eastx - \m}, 0);
  \end{tikzpicture}%
  \par
  \nointerlineskip
  \noindent
}

\newcommand{\DrawSidebarVerticalLine}{
  \begin{tikzpicture}[overlay, remember picture]
    \path (current page.west);
    \pgfgetlastxy{\westx}{\dummy};
    \path (0,0);
    \pgfgetlastxy{\dummyx}{\currenty}
    \path (current page.south);
    \pgfgetlastxy{\dummyx}{\southy}
    \path (current page.north);
    \pgfgetlastxy{\dummyx}{\northy}
    \pgfmathsetlengthmacro{\w}{\SidebarWidth}
    \pgfmathsetlengthmacro{\m}{\PageMargin}
    \pgfmathsetlengthmacro{\l}{\m + \westx + \w}
    \draw (\l, {\southy - \currenty + \m}) -- (\l, {\northy + \currenty - \m});
  \end{tikzpicture}%
}
