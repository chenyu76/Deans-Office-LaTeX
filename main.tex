\newlength{\TableBlockSkip}
\setlength{\TableBlockSkip}{0.5em}

% 自动间距状态：0=初始，1=刚画完横线/表格边界，2=刚写完正文
\newcount\TableAutoSpacingState
\TableAutoSpacingState=0
\newif\ifTableAutoSpacingActive
\newif\ifTableAutoSpacingSuppressPara

\newcommand{\TableAutoSpacingReset}{%
  \global\TableAutoSpacingState=0\relax
}
\newcommand{\TableAutoSpacingMarkRule}{%
  \global\TableAutoSpacingState=1\relax
}
\newcommand{\TableAutoSpacingMarkText}{%
  \global\TableAutoSpacingState=2\relax
}

% 普通正文段落紧跟横线/表格边界时，自动补正文前间距。
\AddToHook{para/begin}{%
  \ifTableAutoSpacingActive
    \ifTableAutoSpacingSuppressPara
    \else
      \ifnum\TableAutoSpacingState=1
        \vadjust pre{\vskip\TableBlockSkip}%
      \fi
      \TableAutoSpacingMarkText
    \fi
  \fi
}

% 横线/表格边界紧跟普通正文时，自动补正文后间距。
\newcommand{\TableAutoSpacingBeforeRule}{%
  \ifTableAutoSpacingActive
    \ifnum\TableAutoSpacingState=2
      \par
      \addvspace{\TableBlockSkip}%
    \fi
  \fi
}

% 命令：绘制横向分割线（底层命令，只画线，不自动加间距）
\newcommand{\RawSepLine}{%
  \begingroup
  \TableAutoSpacingSuppressParatrue
  \par
  \nointerlineskip % 防止垂直间距干扰
  \noindent
  \begin{tikzpicture}[overlay, remember picture]
    \pgfmathsetlengthmacro{\m}{\PageMargin}
    % \pgfmathsetmacro{\h}{-1};
    % \path (current page.center);
    % \pgfgetlastxy{\centerx}{\centery}; % 提取 x 和 y 坐标
    \path (current page.west);
    \pgfgetlastxy{\westx}{\westy}
    \path (current page.east);
    \pgfgetlastxy{\eastx}{\easty}
    \draw ({\westx + \m + \SidebarWidth}, 0) -- ({\eastx - \m -
    \SidebarWidthR}, 0);
  \end{tikzpicture}
  \par
  \nointerlineskip
  \endgroup
}

% 命令：绘制横向分割线（状态感知）
\newcommand{\SepLine}{%
  \TableAutoSpacingBeforeRule
  \RawSepLine
  \TableAutoSpacingMarkRule
}

% 绘制带有垂直间距的分隔线
\newcommand{\vspaceSepLine}{
  \ifTableAutoSpacingActive
    \SepLine
  \else
    \vspace{\TableBlockSkip}
    \SepLine
    \vspace{\TableBlockSkip}
  \fi
}

% 命令：多列行
% 用法: \TableRow[最小高度]{列格式}{内容1 & 内容2...}
% 示例: \TableRow{X|c|X}{ 左边 & 中间 & 右边 }
% 注意：列格式不需要写两边的竖线，系统会自动处理
\NewDocumentCommand{\TableRow}{O{\RowHeight} m m}{%
  \SepLine % 一条分割线
  % 使用 tabularx 占满行宽
  % @{} 消除左右默认间距，使表格紧贴版心边缘
  \begingroup
  \TableAutoSpacingSuppressParatrue
  \noindent
  \begin{tabularx}{\textwidth}{@{} #2 @{}}
    % 在每个单元格内插入一个不可见的rule撑开高度
    \rule{0pt}{#1} #3
  \end{tabularx}%
  \endgroup
  \par
  \nointerlineskip % 紧接着画线或下一行
  \SepLine % 每行结束后自动画分割线
}

\textbf{自动间距压力测试。}

以下内容用于检查：连续表格行之间不应出现额外空白；正文与横线、
正文与表格行之间应自动产生常规间距。

\begin{TableFrame}
  \TableRow{l}{\textbf{测试一：连续 \texttt{\textbackslash{}TableRow}}}
  \TableRow{X|X}{连续行 A 左 & 连续行 A 右}
  \TableRow{X|X}{连续行 B 左 & 连续行 B 右}
  \TableRow{X|X}{连续行 C 左 & 连续行 C 右}

  这段正文紧跟连续表格行。这里不手动写
  \texttt{\textbackslash{}vspace\{0.5em\}}，用于检查
  \texttt{rule -> text} 是否自动补出正文前间距。

  这段正文后面马上接新的表格行，用于检查
  \texttt{text -> rule} 是否只补一次间距。

  \TableRow{l}{\textbf{测试二：正文后接表格行}}
  \TableRow{c|X|c|X}{
    甲 & 正文后第一行 & 乙 & 正文后第二列
  }

  这段正文后面接一条状态感知分隔线。

  \vspaceSepLine

  分隔线后面直接写正文，用于检查分隔线上下的常规间距。

  \vspaceSepLine
  \vspaceSepLine

  上面连续调用了两次 \texttt{\textbackslash{}vspaceSepLine}；
  两条线之间不应叠加出多余的 \texttt{0.5em} 空白。

  \TableRow{l}{\textbf{测试三：正文、公式、列表混排}}

  这里先写普通正文，然后接列表：
  \begin{itemize}
    \item 列表项一：检查列表环境不会破坏后续表格行间距。
    \item 列表项二：这里包含行内公式 $a^2+b^2=c^2$。
  \end{itemize}

  列表后接显示公式：
  \begin{equation}
    \int_0^1 x^2\,dx = \frac{1}{3}

  是否自动补底部安全间距，避免文字和底部边框重叠。
\end{TableFrame}
