  \fi

  \fi
  \fi

% 命令：绘制横向分割线（底层命令，只画线，不自动加间距）
\newcommand{\RawSepLine}{%
  \begingroup
  \TableAutoSpacingSuppressParatrue
  \par
  \nointerlineskip % 防止垂直间距干扰
  \noindent
  \begin{tikzpicture}[overlay, remember picture]
    \pgfmathsetlengthmacro{\m}{\PageMargin}
    \path (current page.west);
    \pgfgetlastxy{\westx}{\westy}
    \path (current page.east);
    \pgfgetlastxy{\eastx}{\easty}
    \draw ({\westx + \m + \SidebarWidth}, 0) -- ({\eastx - \m -
    \SidebarWidthR}, 0);
  \end{tikzpicture}
  \par
  \nointerlineskip
  \endgroup
}

\newcommand{\FullRawSepLine}{%
  \begingroup
  \TableAutoSpacingSuppressParatrue
  \par
  \nointerlineskip
  \noindent
  \begin{tikzpicture}[overlay, remember picture]
    \pgfmathsetlengthmacro{\m}{\PageMargin}
    \path (current page.west);
    \pgfgetlastxy{\westx}{\westy}
    \path (current page.east);
    \pgfgetlastxy{\eastx}{\easty}
    \draw ({\westx + \m}, 0) -- ({\eastx - \m}, 0);
  \end{tikzpicture}
  \par
  \nointerlineskip
  \endgroup
}

% 命令：绘制横向分割线（状态感知）
\newcommand{\SepLine}{%
  \ifTableSkipNextSepLine
  \global\TableSkipNextSepLinefalse
  \else
  \TableAutoSpacingBeforeRule
  \RawSepLine
  \fi
  \TableAutoSpacingMarkRule
}

\newcommand{\SetupSidebarSide}[2]{%
  \if L#1
  \edef\SidebarSideAnchor{south west}%
  \edef\SidebarSideNAnchor{north west}%
  \edef\SidebarSideLineShift{\dimexpr\PageMargin+#2\relax}%
  \edef\SidebarSideTitleShift{\dimexpr\PageMargin+#2/2\relax}%
  \else
  \edef\SidebarSideAnchor{south east}%
  \edef\SidebarSideNAnchor{north east}%
  \edef\SidebarSideLineShift{\dimexpr-\PageMargin-#2\relax}%
  \edef\SidebarSideTitleShift{\dimexpr-\PageMargin-#2/2\relax}%
  \fi
}

\NewDocumentCommand{\DrawSidebarSegment}{m m m m +m}{%
  \begin{tikzpicture}[overlay, remember picture]
    \SetupSidebarSide{#1}{#4}%
    \coordinate (sideLineStart) at
    ([xshift=\SidebarSideLineShift,
    yshift={\dimexpr\zposy{#2}sp\relax}]current page.\SidebarSideAnchor);
    \coordinate (sideLineEnd) at
    ([xshift=\SidebarSideLineShift,
    yshift={\dimexpr\zposy{#3}sp\relax}]current page.\SidebarSideAnchor);
    \coordinate (sideTitleStart) at
    ([xshift=\SidebarSideTitleShift,
    yshift={\dimexpr\zposy{#2}sp\relax}]current page.\SidebarSideAnchor);
    \coordinate (sideTitleEnd) at
    ([xshift=\SidebarSideTitleShift,
    yshift={\dimexpr\zposy{#3}sp\relax}]current page.\SidebarSideAnchor);
    \draw (sideLineStart) -- (sideLineEnd);
    \node at ($(sideTitleStart)!0.5!(sideTitleEnd)$) {
      \begin{minipage}{\dimexpr#4-2\ContentGap\relax}
        #5
      \end{minipage}
    };
  \end{tikzpicture}%
}

% 命令：多列行
% 用法: \TableRow[最小高度]{列格式}{内容1 & 内容2...}
% 示例: \TableRow{X|c|X}{ 左边 & 中间 & 右边 }
% 注意：列格式不需要写两边的竖线，系统会自动处理
\NewDocumentCommand{\TableRow}{O{\RowHeight} m m}{%
  \SepLine % 一条分割线
  % 使用 tabularx 占满行宽
  % @{} 消除左右默认间距，使表格紧贴版心边缘
  \begingroup
  \TableAutoSpacingSuppressParatrue
  \noindent
  \begin{tabularx}{\textwidth}{@{} #2 @{}}
    % 在每个单元格内插入一个不可见的rule撑开高度
    \rule{0pt}{#1} #3
  \end{tabularx}%
  \endgroup
  \par
  \nointerlineskip % 紧接着画线或下一行
  \SepLine % 每行结束后自动画分割线
}

\newcommand{\DrawSidebarHeadFrom}[2]{%
  \begin{tikzpicture}[overlay, remember picture]
    \if L#1
    \edef\SidebarLineWidth{\the\SidebarWidthSaved}%
    \let\SidebarTitle\SideFirstTitleSaved
    \else
    \edef\SidebarLineWidth{\the\SidebarWidthSavedR}%
    \let\SidebarTitle\SideFirstTitleSavedR
    \fi
    \SetupSidebarSide{#1}{\SidebarLineWidth}%
    \coordinate (sideLineStart) at
    ([xshift=\SidebarSideLineShift,
    yshift={\dimexpr\zposy{#2}sp\relax}]current page.\SidebarSideAnchor);
    \coordinate (sideLineBottom) at
    ([xshift=\SidebarSideLineShift,
    yshift=\PageMargin]current page.\SidebarSideAnchor);
    \coordinate (sideTitleStart) at
    ([xshift=\SidebarSideTitleShift,
    yshift={\dimexpr\zposy{#2}sp\relax}]current page.\SidebarSideAnchor);
    \coordinate (sideTitleBottom) at
    ([xshift=\SidebarSideTitleShift,
    yshift=\PageMargin]current page.\SidebarSideAnchor);
    \draw (sideLineStart) -- (sideLineBottom);
    \node at ($(sideTitleStart)!0.5!(sideTitleBottom)$) {
      \begin{minipage}{\dimexpr\SidebarLineWidth-2\ContentGap\relax}
        \SidebarTitle
      \end{minipage}
    };
  \end{tikzpicture}%
}

    由于我们使用的是 overlay 画外框，这里的文本可

    \begin{equation}

    \end{equation}

    或者插入一个普通的表格：

    \begin{table}[H] % 这个 H 标记控制表格不浮动
      \caption{几个不同国家的人均人口}
      \begin{center}
        \begin{tabular}[c]{|l|r|}
          \hline
          国家 & 人均人口 \\
          \hline
          中国 & 1 \\
          美国 & 1 \\
          俄罗斯 & 1 \\
          日本 & 1 \\
          \hline
        \end{tabular}
      \end{center}
    \end{table}

    \begin{figure}[H]
      \begin{center}
        \includegraphics[width=0.5\textwidth]{example-image-a}

      \caption{示例图片}
    \end{figure}
